\section{ناسازگاری شناسهٔ شرکت‌کننده}

\begin{apibox}
\apipath{DELETE /sessions/<session\_id>/participants/<participant\_id>}
\end{apibox}

\begin{problembox}
در فهرست مسیرها \lr{participant\_id} آمده؛ در توضیح همان \lr{endpoint} نوشته شده \lr{user\_id}.

\end{problembox}

\begin{solutionbox}

\apipath{DELETE /sessions/<session\_id>/participants/<user\_id>}
\end{solutionbox}

\section{ساخت آزمون تک‌مرحله‌ای}

\begin{apibox}
\apipath{POST /exams}
\end{apibox}

\begin{problembox}
\begin{enumerate}
  \item با دامنهٔ فعلی هم‌خوان نیست. جریان امروز
        \lr{Problem → Exam.add\_problem → publish}
        است؛ این \lr{endpoint} یک \lr{orchestration} یک‌مرحله‌ای است که هنوز \lr{Service} ندارد.
  \item فیلد \lr{source} بدون \lr{order\_index} آمده؛ در
        \lr{SinglePDFSourceSchema} این فیلد اجباری است.
  \item \lr{exam\_type} و \lr{config.render\_mode} تکراری‌اند.
  \item \lr{score} و \lr{negative\_score} مشخص نیست؛ دامنه \lr{Scoring} اجباری دارد.
  \item مشخص نیست خروجی \lr{create} آزمون را \lr{PUBLISHED} می‌کند یا \lr{DRAFT}؛
        در حالی که \lr{session} فقط از آزمون منتشرشده ساخته می‌شود.
\end{enumerate}
\end{problembox}

\begin{solutionbox}
\begin{itemize}
  \item یک \lr{CreateSinglePdfExamService} تعریف کنید که زیرپوست:
        $N$ سؤال \lr{Problem(MCQ)} بسازد، به \lr{Exam} وصل کند، \lr{publish} کند و
        \lr{exam\_id} برگرداند.
  \item برای منبع \lr{PDF} سطح آزمون، \lr{schema} جدا بدون \lr{order\_index} بگذارید؛
        \lr{order\_index} فقط روی \lr{meta} هر سؤال بماند.
  \item یکی از \lr{exam\_type} یا \lr{render\_mode} را حذف کنید.
  \item حداقل یک \lr{default} سراسری برای نمرهٔ مثبت و منفی در \lr{body} یا سمت سرور مشخص کنید.
  \item رفتار را صریح بنویسید: \lr{POST /exams = create + publish}.
\end{itemize}
\end{solutionbox}

\section{ساخت جلسه و مفهوم اسکوپ و پرایوت}

\begin{apibox}
\noindent مسیر: \apipath{POST /exams/<exam\_id>/sessions}\\
بخش متنی مورد ایراد: جملهٔ «آزمون پرایوت / اسکوپ»
\end{apibox}

\begin{problembox}
در پیش‌نویس آمده اگر آزمون \lr{private} باشد فقط \lr{scope} مربوط می‌تواند \lr{session} بسازد.
در Athena نه \lr{private} داریم نه \lr{school/scope}. Tenancy متعلق به Desko است.
\end{problembox}

\begin{solutionbox}
این جمله را از قرارداد Athena حذف کنید.
احراز مالکیت و مجوز ساخت جلسه در \lr{Gateway}/\lr{Desko} انجام شود،
یا حداکثر با فیلد \lr{created\_by} روی \lr{Exam}.
\end{solutionbox}

\section{پاسخ افزودن شرکت‌کنندگان}

\begin{apibox}
\apipath{POST /sessions/<session\_id>/participants}
\end{apibox}

شکل \lr{Response} با علامت سؤال رها شده و قفل نیست.

\end{itemize}
\end{solutionbox}

\begin{codebox}
\codeline{added: list[uuid]}
\apipath{already\_assigned: list[uuid]}

\section{سؤالات مدیر و افشای کلید}

\begin{apibox}
\noindent مسیر: \apipath{GET /sessions/<session\_id>/problems}\\
مخاطب اعلام‌شده در پیش‌نویس: مدیر (\lr{manager})
\end{apibox}

\begin{problembox}
پاسخ شامل \lr{private\_payload} (کلید جواب) است، اما قانون دسترسی فقط با عبارت
\lr{for manager} آمده. در صورت افشای اشتباه، کلید لو می‌رود.
\end{problembox}

\begin{solutionbox}
در قرارداد صریح بنویسید: فقط سازنده یا مدیر همان آزمون یا جلسه؛ در غیر این صورت \lr{404}.
هرگز این داده را از مسیرهای \lr{attempt} برنگردانید.
\end{solutionbox}

\section{خطاهای شروع Attempt}

\begin{apibox}
\noindent مسیر: \apipath{POST /sessions/<session\_id>/attempts}\\
بخش مورد ایراد: خطاهای \lr{409}
\end{apibox}

\begin{problembox}
کد \lr{409 session has not started} با رفتار فعلی \lr{StartAttemptService} ممکن است نخواند.
آن سرویس عمدتاً روی بازهٔ زمانی کار می‌کند، نه لزوماً روی \lr{status = RUNNING}.
\end{problembox}

\begin{solutionbox}
یکی را انتخاب کنید و در API و دامنه یکسان کنید:
\begin{itemize}
  \item اعمال قاعده روی \lr{start\_time}/\lr{end\_time}
  \item یا اعمال قاعده روی \lr{status}
\end{itemize}
همان قاعده را در فهرست \lr{Errors} بنویسید.
وضعیت‌های \lr{cancelled}، \lr{expired} و \lr{already submitted} را نگه دارید.
\end{solutionbox}

\section{یکدستی کدهای خطای دسترسی}

\begin{apibox}
بخش مورد بررسی: کدهای خطای \lr{ownership} و \lr{participant}
\end{apibox}

\begin{problembox}
گاهی برای \lr{user\_id ≠ sub} کد \lr{404} آمده، گاهی برای participant تخصیص‌نیافته هم \lr{404}،
و در عمل مفهومی شبیه \lr{403} هم مطرح است. قرارداد یکدست نیست.
\end{problembox}

\begin{solutionbox}
قرارداد پیشنهادی:
\begin{itemize}
  \item منبع واقعاً وجود ندارد ← \lr{404}
  \item \lr{attempt} متعلق به این کاربر نیست ← \lr{404} (ضد شمارش)
  \item کاربر در \lr{roster} نیست ← \lr{403} با
        \lr{error\_code = participant\_is\_not\_assigned}
  \item تعارض وضعیت ← \lr{409}
\end{itemize}
\end{solutionbox}

\section{نتیجهٔ Attempt تعریف‌نشده}

\begin{apibox}
\apipath{GET /attempts/<attempt\_id>/result}
\end{apibox}

\begin{problembox}
فقط با علامت سؤال رها شده است. بدون قرارداد، \lr{query} و UI بسته نمی‌شود.
ضمناً مسیر \lr{grade → finish} در کد فعلی مشکل دارد؛ این \lr{endpoint} بدون رفع آن بی‌معناست.
\end{problembox}

\begin{solutionbox}
حداقل فیلدهای زیر را برای پاسخ قفل کنید. هم‌زمان باگ‌های \lr{submit}، \lr{grading} و \lr{finish}
را قبل یا همراه همین \lr{endpoint} ببندید.
تصمیم محصولی را هم بنویسید: بعد از \lr{submit} آیا \lr{is\_correct} یا کلید نشان داده شود، یا فقط نمرهٔ کل؟
\end{solutionbox}

\begin{codebox}
\codeline{attempt\_id: uuid}
\codeline{status: FINISHED}
\codeline{score / max\_score / min\_score: decimal string}
\codeline{submitted\_at: ISO-8601}
\codeline{answers[].exam\_session\_problem\_id: uuid}
\codeline{answers[].selected\_option\_id: int}
\codeline{answers[].score: decimal string}
\apipath{answers[].is\_correct: bool}
\end{codebox}

\section{محدوده نسبت به MVP محصول}

\begin{apibox}
بخش مورد بررسی: کل قرارداد API نسبت به استوری‌های فاز ۱ سی‌ای‌او
\end{apibox}

\begin{problembox}
این API تقریباً فقط برگزاری مجازی تستی و تصحیح خودکار \lr{MCQ} را می‌پوشاند.
در استوری‌های فاز ۱ محصول این موارد هم هست و اینجا نیست:
\begin{itemize}
  \item سؤال تشریحی و تنظیم نوع پاسخ
  \item اعتراض به نمره
  \item رویداد ضدتقلب (\lr{focus lost})
  \item مصحح و اکانت مهمان
  \item \lr{OCR} و آزمون حضوری
\end{itemize}
اگر بدون اعلام محدوده جلو برویم، ظرفیت سه هفته روی انتظار اشتباه مصرف می‌شود.
\end{problembox}

\begin{solutionbox}
بالای \lr{api.md} صریحاً بنویسید:

\begin{quote}
\textbf{محدودهٔ این قرارداد:} آزمون مجازی \lr{SINGLE\_PDF} به‌همراه \lr{MCQ}، \lr{session}، \lr{attempt} و \lr{auto-grade}.\\
\textbf{خارج از محدوده:} تشریحی، \lr{OCR} فیزیکی، مصحح، مهمان، اعتراض، ضدتقلب پیشرفته.
\end{quote}

اگر محصول روی ضدتقلب پایه در همین فاز اصرار دارد، حداقل این مسیر را اضافه کنید:
\end{solutionbox}

\begin{codebox}
\codeline{POST /attempts/<attempt\_id>/events}
\codeline{type: FOCUS\_LOST}
\codeline{occurred\_at: datetime}
\apipath{meta: object}
\end{codebox}
